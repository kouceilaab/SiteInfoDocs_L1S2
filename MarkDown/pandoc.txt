#La validation xHTML1.0 ne passait pas avec markdwon donc j'ai utilisé commonmark en ajoutant l'option des identifiants automatiques et les taleaux



pandoc -f commonmark+gfm_auto_identifiers+pipe_tables -t html -s 
--template template_Bechouche_Boukkerouche.html 
--metadata-file variables.yml 
--metadata pagetitle="Accueil" 
--metadata idPage="accueil" index.md -o ../HTML/index.html
